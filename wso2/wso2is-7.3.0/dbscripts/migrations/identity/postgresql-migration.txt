#This contains the database changes required to apply token binding support for non-persistent access token scenarios.

CREATE TABLE IF NOT EXISTS IDN_OAUTH2_REFRESH_TOKEN_BINDING (
    REFRESH_TOKEN_ID VARCHAR (255) NOT NULL,
    TOKEN_BINDING_TYPE VARCHAR (32) NOT NULL,
    TOKEN_BINDING_REF VARCHAR (32) NOT NULL,
    TOKEN_BINDING_VALUE VARCHAR (1024) NOT NULL,
    TENANT_ID INTEGER DEFAULT -1,
    PRIMARY KEY (REFRESH_TOKEN_ID, TOKEN_BINDING_TYPE),
    UNIQUE (REFRESH_TOKEN_ID, TOKEN_BINDING_TYPE, TOKEN_BINDING_VALUE),
    FOREIGN KEY (REFRESH_TOKEN_ID) REFERENCES IDN_OAUTH2_REFRESH_TOKEN(REFRESH_TOKEN_ID) ON DELETE CASCADE
);

CREATE INDEX IF NOT EXISTS IDX_REFRESH_TOKEN_BIND ON IDN_OAUTH2_REFRESH_TOKEN_BINDING (TOKEN_BINDING_REF);
CREATE INDEX IF NOT EXISTS IDX_REFRESH_TK_VALUE_TYPE ON IDN_OAUTH2_REFRESH_TOKEN_BINDING (TOKEN_BINDING_VALUE, TOKEN_BINDING_TYPE);
