#update 18910

#This contains the database changes required for the consent management feature.

ALTER TABLE CM_PII_CATEGORY ADD COLUMN UUID CHAR(36);

ALTER TABLE CM_RECEIPT ADD COLUMN CURSOR_KEY INTEGER NOT NULL AUTO_INCREMENT;
ALTER TABLE CM_RECEIPT ADD UNIQUE KEY (CURSOR_KEY);
ALTER TABLE CM_RECEIPT ADD COLUMN EXPIRY_TIME TIMESTAMP NULL;

CREATE TABLE IF NOT EXISTS CM_CONSENT_AUTHORIZATION (
  CONSENT_RECEIPT_ID VARCHAR(255) NOT NULL,
  USER_ID            VARCHAR(255) NOT NULL,
  STATUS             VARCHAR(50)  NOT NULL,
  UPDATED_TIME       BIGINT       NOT NULL,
  TYPE               VARCHAR(255),
  PRIMARY KEY (CONSENT_RECEIPT_ID, USER_ID)
) ENGINE NDB;

ALTER TABLE CM_PURPOSE ADD COLUMN UUID CHAR(36);
ALTER TABLE CM_PURPOSE ADD COLUMN LATEST_VERSION_ID CHAR(36);

ALTER TABLE CM_SP_PURPOSE_ASSOC ADD COLUMN PURPOSE_VERSION_ID CHAR(36);

CREATE TABLE IF NOT EXISTS CM_PURPOSE_VERSION (
  UUID CHAR(36) NOT NULL,
  PURPOSE_ID  INTEGER NOT NULL,
  VERSION     VARCHAR(100) NOT NULL,
  DESCRIPTION VARCHAR(1023),
  TENANT_ID   INTEGER DEFAULT '-1234',
  UNIQUE KEY (PURPOSE_ID, VERSION),
  UNIQUE KEY (PURPOSE_ID, UUID),
  PRIMARY KEY (TENANT_ID, UUID)
) ENGINE NDB;

CREATE TABLE IF NOT EXISTS CM_PURPOSE_VERSION_PII_CAT_ASSOC (
  PURPOSE_VERSION_ID CHAR(36) NOT NULL,
  CM_PII_CATEGORY_ID INTEGER NOT NULL,
  IS_MANDATORY       INTEGER NOT NULL,
  TENANT_ID          INTEGER NOT NULL DEFAULT -1234,
  PRIMARY KEY (PURPOSE_VERSION_ID, CM_PII_CATEGORY_ID)
) ENGINE NDB;

CREATE TABLE IF NOT EXISTS CM_PURPOSE_VERSION_PROPERTY (
  VERSION_ID     CHAR(36)      NOT NULL,
  PROPERTY_KEY   VARCHAR(255)  NOT NULL,
  PROPERTY_VALUE VARCHAR(1023) NOT NULL,
  TENANT_ID      INTEGER       NOT NULL DEFAULT -1234,
  PRIMARY KEY (VERSION_ID, PROPERTY_KEY)
) ENGINE NDB;

CREATE TABLE IF NOT EXISTS CM_PII_CATEGORY_PROPERTY (
  PII_CATEGORY_ID  INTEGER       NOT NULL,
  PROPERTY_KEY     VARCHAR(255)  NOT NULL,
  PROPERTY_VALUE   VARCHAR(1023) NOT NULL,
  TENANT_ID        INTEGER       NOT NULL DEFAULT -1234,
  PRIMARY KEY (PII_CATEGORY_ID, PROPERTY_KEY),
  FOREIGN KEY (PII_CATEGORY_ID) REFERENCES CM_PII_CATEGORY(ID) ON DELETE CASCADE
) ENGINE NDB;

ALTER TABLE CM_PURPOSE_VERSION_PROPERTY
  ADD CONSTRAINT CM_PV_PROPERTY_fk0 FOREIGN KEY (TENANT_ID, VERSION_ID) REFERENCES CM_PURPOSE_VERSION (TENANT_ID, UUID) ON DELETE CASCADE;

ALTER TABLE CM_CONSENT_AUTHORIZATION
  ADD CONSTRAINT CM_CONSENT_AUTH_fk0 FOREIGN KEY (CONSENT_RECEIPT_ID) REFERENCES CM_RECEIPT (CONSENT_RECEIPT_ID) ON DELETE CASCADE;

ALTER TABLE CM_SP_PURPOSE_ASSOC
  ADD CONSTRAINT CM_SP_PURPOSE_ASSOC_fk2 FOREIGN KEY (PURPOSE_ID, PURPOSE_VERSION_ID) REFERENCES CM_PURPOSE_VERSION (PURPOSE_ID, UUID) ON DELETE CASCADE;

ALTER TABLE CM_PURPOSE_VERSION
  ADD CONSTRAINT CM_PURPOSE_VERSION_fk0 FOREIGN KEY (PURPOSE_ID) REFERENCES CM_PURPOSE (ID) ON DELETE CASCADE;

ALTER TABLE CM_PURPOSE
  ADD CONSTRAINT CM_PURPOSE_LATEST_VER_fk FOREIGN KEY (ID, LATEST_VERSION_ID) REFERENCES CM_PURPOSE_VERSION (PURPOSE_ID, UUID);

ALTER TABLE CM_PURPOSE_VERSION_PII_CAT_ASSOC
  ADD CONSTRAINT CM_PV_PII_CAT_ASSOC_fk0 FOREIGN KEY (TENANT_ID, PURPOSE_VERSION_ID) REFERENCES CM_PURPOSE_VERSION (TENANT_ID, UUID) ON DELETE CASCADE;

ALTER TABLE CM_PURPOSE_VERSION_PII_CAT_ASSOC
  ADD CONSTRAINT CM_PV_PII_CAT_ASSOC_fk1 FOREIGN KEY (CM_PII_CATEGORY_ID) REFERENCES CM_PII_CATEGORY (ID) ON DELETE CASCADE;

ALTER TABLE CM_PII_CATEGORY ADD INDEX CM_PII_CATEGORY_UUID_IDX (UUID) ENGINE=NDDB;
ALTER TABLE CM_PURPOSE ADD INDEX CM_PURPOSE_UUID_IDX (UUID) ENGINE=NDDB;
